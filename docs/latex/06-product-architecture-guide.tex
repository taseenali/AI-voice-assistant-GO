\section{What This Product Is}

\subsection{One-Sentence Pitch}
\textbf{MedVoice AI is a 24/7 AI medical receptionist sold to clinics as a subscription.} It answers phone calls and website chats, collects patient information, books appointments into Google Calendar, escalates emergencies, and surfaces everything in a clinic dashboard.

\subsection{The Analogy}
You sell two things:
\begin{enumerate}[nosep]
  \item \textbf{The product} --- prompts, medical safety, booking logic, dashboard, billing. \emph{This is MedVoice.}
  \item \textbf{The voice engine} --- real-time phone audio via Vapi + Twilio. \emph{Rented infrastructure.}
\end{enumerate}
Clinics care that calls get answered and appointments land on their calendar --- not which runtime you use.

\subsection{Who Buys It}
\begin{center}
\small
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Buyer} & \textbf{Pain} \\
\midrule
Solo doctor / small clinic & Missed after-hours calls; no budget for 2 FTE staff \\
Dental / urgent care & High volume; repetitive booking questions \\
Multi-provider group & Inconsistent front desk; staff turnover \\
\bottomrule
\end{tabular}
\end{center}

\subsection{Problems Solved}
\begin{itemize}[nosep]
  \item Missed calls $\rightarrow$ lost appointments (\$150--300 each)
  \item After-hours voicemail $\rightarrow$ patients call the next clinic
  \item Staff cost: $\sim$\$9K/mo (2 receptionists) vs.\ \$1.5--3K/mo (MedVoice)
  \item Emergency risk: hard-coded detector bypasses LLM for red-flag keywords
\end{itemize}

\subsection{Web Chat Flow (Functional)}
\begin{enumerate}[nosep]
  \item Patient opens widget (\texttt{?client=medical-clinic})
  \item Widget loads config from \texttt{GET /api/config}
  \item \textbf{ResponseOrchestrator} runs 6-step pipeline per message
  \item Emergency detector runs first (keyword match, bypasses LLM)
  \item Lead capture collects name, DOB, insurance, reason
  \item Turns logged to SQLite; webhooks fire to clinic n8n URL
  \item Clinic staff views data in dashboard
\end{enumerate}

\section{Database: Current vs.\ Target}

\subsection{Existing Tables (SQLite)}
\begin{center}
\small
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Table} & \textbf{Purpose} \\
\midrule
\texttt{sessions} & One conversation; \texttt{client\_id}, duration, flags \\
\texttt{conversation\_turns} & Each message; role, intent, state \\
\texttt{leads} & Patient info: name, DOB, insurance, reason \\
\texttt{emergency\_events} & Red-flag keyword detections \\
\texttt{audit\_logs} & HIPAA trail (exists; barely used) \\
\bottomrule
\end{tabular}
\end{center}

Config lives in \texttt{configs/*.json}, not the database. No \texttt{users}, \texttt{tenants}, or \texttt{appointments} tables yet.

\subsection{Target Schema (Production SaaS)}

\begin{center}
\small
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{New Table} & \textbf{Purpose} \\
\midrule
\texttt{tenants} & Company, plan, Stripe ID, status \\
\texttt{users} & Email, role (super\_admin / clinic\_admin / staff) \\
\texttt{tenant\_config} & Prompt, voice, hours, services (replaces JSON) \\
\texttt{phone\_numbers} & E.164 $\rightarrow$ tenant mapping \\
\texttt{appointments} & Booked slots, Google event ID \\
\texttt{usage\_records} & Minutes per tenant per month (billing) \\
\texttt{webhook\_deliveries} & Outbound event status, retries \\
\texttt{patients} & Returning patient lookup (Phase 2) \\
\bottomrule
\end{tabular}
\end{center}

\subsection{Database Rules}
\begin{itemize}[nosep]
  \item Every query filtered by \texttt{tenant\_id}
  \item PHI encrypted at rest in production
  \item Audit log on every PHI read/write
  \item Sessions immutable after close
  \item PostgreSQL migration before 10+ clinics
\end{itemize}

\section{Pre-Code Checklist}

\subsection{Phase 0 --- Align (Now)}
\begin{itemize}[nosep]
  \item[$\square$] Product definition agreed
  \item[$\square$] Pick first pilot clinic (number, calendar, services)
  \item[$\square$] Open accounts: Vapi, Twilio, Deepgram, OpenAI, ElevenLabs, GCP
\end{itemize}

\subsection{Phase 1 --- Phone MVP ($\sim$2 weeks)}
\begin{enumerate}[nosep]
  \item \texttt{server/routes/vapi.js} --- assistant-request, tool-calls, end-of-call
  \item Phone number $\rightarrow$ \texttt{client\_id} mapping
  \item Wire Vapi tools $\rightarrow$ calendar, leads, emergency routes
  \item Server-side webhook worker (replace browser outbox for phone)
  \item Enable calendar; E2E booking test
  \item Store \texttt{recording\_url}, \texttt{channel: phone} on sessions
\end{enumerate}
\textbf{Exit:} Call $\rightarrow$ book $\rightarrow$ transcript in dashboard in 30s.

\subsection{Phase 2 --- Clinic Dashboard v1}
\begin{enumerate}[nosep,start=7]
  \item \texttt{users} + \texttt{tenants} tables; JWT auth
  \item API routes require auth + tenant from token
  \item Appointments panel; call recording playback
  \item Editable config (greeting, hours, services)
  \item Emergency email/push notification
  \item Move system health internals to super-admin only
\end{enumerate}

\subsection{Phase 3 --- First Paying Client ($\sim$weeks 3--6)}
\begin{enumerate}[nosep,start=13]
  \item HIPAA hosting + BAAs (all providers)
  \item Wire \texttt{audit\_logs}; encrypt PHI at rest
  \item Stripe tiers + \texttt{usage\_records}
  \item Production deploy; after-hours flow; call transfer; SMS confirm
\end{enumerate}
\textbf{Exit:} One clinic pays \$999--1,499/mo; signed BAA; 2 weeks stable.

\subsection{Phase 4 --- Multi-Tenant SaaS (months 2--4)}
\begin{enumerate}[nosep,start=19]
  \item Super-admin dashboard (\texttt{/admin/*})
  \item Tenant onboarding wizard
  \item Config JSON $\rightarrow$ database
  \item PostgreSQL; reschedule/cancel tools; CI/CD; monitoring
\end{enumerate}
\textbf{Exit:} New clinic live in $<$2 hrs; 10 paying tenants.

\subsection{Phase 5 --- Scale (months 4--18)}
Analytics, outbound campaigns, CRM connectors, EHR (Epic/athena), patient DB, white-label, LiveKit cost migration, SOC 2.

\section{60-Second Pitch}

\emph{``MedVoice is an AI receptionist built for medical clinics. When a patient calls or chats, our assistant Aria answers immediately --- even at 2am --- collects their information, checks the clinic's Google Calendar, and books the appointment. If someone says `chest pain,' we use a hard-coded emergency system that logs the event and offers transfer to staff. Everything appears in a dashboard the office manager uses each morning: new leads, transcripts, and emergency alerts. Clinics pay \$1,500--3,000/month --- less than half one receptionist --- and never miss an after-hours call.''}
