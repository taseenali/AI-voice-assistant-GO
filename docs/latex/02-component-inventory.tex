\section{Missing Components (Must Build)}

\begin{enumerate}
  \item \textbf{Vapi integration layer} --- assistant-request, tool-calls, end-of-call-report
  \item \textbf{Server-side webhook worker} --- replace browser-only delivery for phone
  \item \textbf{Tenant management API + UI} --- onboard clinics without editing JSON
  \item \textbf{Authentication / RBAC} --- protect dashboard and admin APIs
  \item \textbf{Billing (Stripe)} --- subscription tiers, usage metering
  \item \textbf{HIPAA stack} --- encrypted PHI store, audit logs, BAAs, HIPAA-eligible hosting
  \item \textbf{Call transfer / escalation} --- Twilio transfer from Vapi tool
  \item \textbf{SMS confirmations} --- Twilio SMS post-booking
  \item \textbf{After-hours logic} --- business\_hours in config + flow branching
  \item \textbf{Reschedule / cancel} --- calendar event update/delete tools
\end{enumerate}
