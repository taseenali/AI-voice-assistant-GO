\section{Level 0 --- Current State (Complete)}

\textbf{Duration:} Done (Passes 1--8)

\textbf{Delivered:}
\begin{itemize}[nosep]
  \item Web voice/chat widget with full conversation engine
  \item Multi-tenant config via \texttt{?client=}
  \item Ollama LLM proxy, emergency detection (verified)
  \item Lead capture + HMAC webhooks (browser outbox)
  \item SQLite persistence + React dashboard (wired)
  \item Calendar API code (disabled in config)
\end{itemize}

\textbf{Blockers to L1:} No live phone; no Vapi; calendar not E2E verified.

\textbf{Exit criteria:}
\begin{enumerate}[nosep]
  \item Inbound call books real appointment on Google Calendar
  \item Transcript appears in dashboard within 30s of hang-up
  \item Emergency keyword triggers log + transfer offer
  \item Zero browser-tab dependency for phone calls
\end{enumerate}

\textbf{Exit criteria:}
\begin{enumerate}[nosep]
  \item Signed BAA with clinic; BAAs with all providers

\end{enumerate}

\section{Level 4 --- Scale (Months 4--8)}

\textbf{Goal:} Differentiated product; reduce churn; expand ARPU.

\begin{itemize}[nosep]
  \item Analytics dashboard (call volume, booking conversion, after-hours capture)
  \item Outbound campaigns (recall, no-show recovery) via Vapi outbound API
  \item CRM connectors (HubSpot, Salesforce native)
  \item athenahealth / eClinicalWorks calendar integration
  \item White-label option (clinic branding on widget + voice)
  \item SLA tier (99.9\% uptime guarantee)
  \item Evaluate LiveKit migration for high-volume cost savings
\end{itemize}

\textbf{Exit criteria:} 50+ clinics; \$25K+ MRR; $<$\$0.20/min blended cost at scale.

---

\section{Level 5 --- Enterprise (Months 8--18)}

\textbf{Goal:} Multi-location groups; EHR-native; compliance audit-ready.

\begin{itemize}[nosep]
  \item Epic FHIR integration (appointments, patient lookup)
  \item SOC 2 Type II + HIPAA audit
  \item Dedicated CSM + custom contracts
  \item Multi-EHR per tenant (group practices)
  \item Insurance verification API integration
  \item Custom voice cloning per clinic (ElevenLabs Pro)
\end{itemize}

\textbf{Exit criteria:} 1 enterprise contract (\$10K+/mo); audit passed.
